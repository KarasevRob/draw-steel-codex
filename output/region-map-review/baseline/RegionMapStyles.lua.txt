local function SubscribeToTheme(element)
    element.data.themeSubscription = ThemeEngine.OnThemeChanged(mod, function()
        if mod.unloaded or not element.valid then return end
        element.styles = ThemeEngine.MergeStyles(BuildStyles())
        element:UpdateStyle()
    end)
end

local function UnsubscribeFromTheme(element)
    if element.data.themeSubscription ~= nil then
        element.data.themeSubscription:Deregister()
        element.data.themeSubscription = nil
    end
end

Styles.Build = BuildStyles
Styles.Subscribe = SubscribeToTheme
Styles.Unsubscribe = UnsubscribeFromTheme
Styles.ready = true
Addon.Styles = Styles
